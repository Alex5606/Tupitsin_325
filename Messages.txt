package com.example.messenger

import android.graphics.BitmapFactory
import android.util.Base64
import android.view.Gravity
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.FrameLayout
import androidx.core.content.ContextCompat
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.ListAdapter
import androidx.recyclerview.widget.RecyclerView
import com.example.messenger.databinding.ItemMessageBinding
import java.text.SimpleDateFormat
import java.util.*

class MessagesAdapter(
    private val currentUserId: String,
    private val onMessageLongClick: (Message) -> Unit,
    private val onImageClick: (String) -> Unit
) : ListAdapter<Message, MessagesAdapter.ViewHolder>(DiffCallback()) {

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): ViewHolder {
        val binding = ItemMessageBinding.inflate(
            LayoutInflater.from(parent.context), parent, false
        )
        return ViewHolder(binding)
    }

    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        holder.bind(getItem(position))
    }

    inner class ViewHolder(
        private val binding: ItemMessageBinding
    ) : RecyclerView.ViewHolder(binding.root) {

        fun bind(message: Message) {
            val isSent = message.senderId == currentUserId

            val params = binding.messageContainer.layoutParams as FrameLayout.LayoutParams

            if (isSent) {
                params.gravity = Gravity.END
                binding.messageContainer.setBackgroundResource(R.drawable.bg_message_sent)
            } else {
                params.gravity = Gravity.START
                binding.messageContainer.setBackgroundResource(R.drawable.bg_message_received)
            }
            binding.messageContainer.layoutParams = params

            when (message.type) {
                MessageType.TEXT -> {
                    binding.tvMessage.visibility = View.VISIBLE
                    binding.ivMessageImage.visibility = View.GONE
                    binding.tvMessage.text = message.text

                    val textColor = if (isSent) R.color.white else R.color.text_primary
                    binding.tvMessage.setTextColor(ContextCompat.getColor(binding.root.context, textColor))
                }
                MessageType.IMAGE -> {
                    binding.tvMessage.visibility = View.GONE
                    binding.ivMessageImage.visibility = View.VISIBLE

                    try {
                        if (message.imageUrl.length > 100) {
                            val imageBytes = Base64.decode(message.imageUrl, Base64.DEFAULT)
                            val bitmap = BitmapFactory.decodeByteArray(imageBytes, 0, imageBytes.size)
                            binding.ivMessageImage.setImageBitmap(bitmap)
                        }
                    } catch (e: Exception) {
                        binding.ivMessageImage.setImageResource(R.drawable.ic_avatar)
                    }

                    binding.ivMessageImage.setOnClickListener {
                        onImageClick(message.imageUrl)
                    }
                }
            }

            binding.tvTime.text = SimpleDateFormat("HH:mm", Locale.getDefault()).format(Date(message.timestamp))

            val timeColor = if (isSent) R.color.white else R.color.text_secondary
            binding.tvTime.setTextColor(ContextCompat.getColor(binding.root.context, timeColor))

            binding.ivStatus.visibility = if (isSent) View.VISIBLE else View.GONE

            binding.messageContainer.setOnLongClickListener {
                onMessageLongClick(message)
                true
            }
        }
    }

    class DiffCallback : DiffUtil.ItemCallback<Message>() {
        override fun areItemsTheSame(old: Message, new: Message) = old.id == new.id
        override fun areContentsTheSame(old: Message, new: Message) = old == new
    }
}з
